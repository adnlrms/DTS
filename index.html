const INITIAL_ADMIN_CODE = "ADMIN-2026";
const INITIAL_OVERRIDE_CODE = "OVERRIDE-2026";

function getAdminCode() {
  const props = PropertiesService.getScriptProperties();
  let code = props.getProperty("ADMIN_SECRET_CODE");
  if (!code) { code = INITIAL_ADMIN_CODE; props.setProperty("ADMIN_SECRET_CODE", code); }
  return code;
}

function getRouteOverrideCode() {
  const props = PropertiesService.getScriptProperties();
  let code = props.getProperty("ROUTE_OVERRIDE_CODE");
  if (!code) { code = INITIAL_OVERRIDE_CODE; props.setProperty("ROUTE_OVERRIDE_CODE", code); }
  return code;
}

function getDatabaseSpreadsheet() {
  let ss = null;
  try { ss = SpreadsheetApp.getActiveSpreadsheet(); } catch (e) {}
  if (ss) return ss;
  const props = PropertiesService.getScriptProperties();
  const savedSheetId = props.getProperty("DATABASE_SHEET_ID");
  if (savedSheetId) { try { ss = SpreadsheetApp.openById(savedSheetId); if (ss) return ss; } catch (e) {} }
  ss = SpreadsheetApp.create("DocTrack QR - DepEd Agusan del Norte Database");
  props.setProperty("DATABASE_SHEET_ID", ss.getId());
  return ss;
}

function getOrCreateArchiveFolder() {
  const props = PropertiesService.getScriptProperties();
  const savedFolderId = props.getProperty("ARCHIVE_FOLDER_ID");
  if (savedFolderId) { try { const f = DriveApp.getFolderById(savedFolderId); if (f) return f; } catch (e) {} }
  const folder = DriveApp.createFolder("DocTrack Completed Document Scans");
  try { folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}
  props.setProperty("ARCHIVE_FOLDER_ID", folder.getId());
  return folder;
}

function getOrCreateMonthlyArchiveFolder() {
  const motherFolder = getOrCreateArchiveFolder();
  const now = new Date();
  const yyyy = now.getFullYear();
  const mm = String(now.getMonth() + 1).padStart(2, "0");
  const monthNames = ["January","February","March","April","May","June","July","August","September","October","November","December"];
  const folderName = yyyy + "-" + mm + " (" + monthNames[now.getMonth()] + " " + yyyy + ")";
  const existing = motherFolder.getFoldersByName(folderName);
  if (existing.hasNext()) { return existing.next(); }
  const monthFolder = motherFolder.createFolder(folderName);
  try { monthFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}
  return monthFolder;
}

function purgeExpiredTrash(docSheet, logSheet) {
  try {
    const now = new Date();
    const currYyyyMm = now.getFullYear() + "-" + String(now.getMonth() + 1).padStart(2, "0");
    const data = docSheet.getDataRange().getValues();
    if (data.length < 2) return;
    const headers = data[0];
    const statusIdx = headers.indexOf("Status");
    const metaIdx = headers.indexOf("MetaJSON");
    const idIdx = headers.indexOf("TrackingID");
    if (statusIdx === -1 || metaIdx === -1 || idIdx === -1) return;

    for (let i = data.length; i >= 2; i--) {
      const row = data[i - 1];
      if (String(row[statusIdx]) === "TRASH") {
        let meta = {};
        try { meta = JSON.parse(row[metaIdx] || "{}"); } catch (e) {}
        const deletedMonth = meta.deletedMonth || "";
        if (deletedMonth && deletedMonth !== currYyyyMm) {
          const tid = String(row[idIdx]);
          docSheet.deleteRow(i);
          // Also delete tracking logs for this purged document
          const lData = logSheet.getDataRange().getValues();
          for (let j = lData.length; j >= 2; j--) {
            if (String(lData[j-1][1]) === tid) { logSheet.deleteRow(j); }
          }
        }
      }
    }
  } catch (e) {}
}

function setupSheets() {
  const ss = getDatabaseSpreadsheet();
  let docSheet = ss.getSheetByName("Documents");
  if (!docSheet) {
    docSheet = ss.insertSheet("Documents");
    docSheet.appendRow(["TrackingID", "Title", "StartParty", "Type", "Priority", "OriginOffice", "DestinationOffice", "EndNote", "Remarks", "Status", "CurrentOffice", "CreatedBy", "CreatedAt", "MetaJSON", "LastUpdatedAt", "ScannedFileUrl"]);
    docSheet.getRange("A1:P1").setFontWeight("bold").setBackground("#E8F0FE");
  }
  let logSheet = ss.getSheetByName("TrackingLogs");
  if (!logSheet) {
    logSheet = ss.insertSheet("TrackingLogs");
    logSheet.appendRow(["LogID", "TrackingID", "Office", "Action", "HandledBy", "Remarks", "Timestamp", "FileUrl"]);
    logSheet.getRange("A1:H1").setFontWeight("bold").setBackground("#E6F4EA");
  }
  let userSheet = ss.getSheetByName("Users");
  const expectedUserHeaders = ["UserID", "Name", "Office", "OfficeCode", "Username", "Password", "Role", "CreatedAt"];
  if (!userSheet) {
    userSheet = ss.insertSheet("Users");
    userSheet.appendRow(expectedUserHeaders);
    userSheet.getRange("A1:H1").setFontWeight("bold").setBackground("#FEF7E0");
    userSheet.appendRow(["USR-01", "Records Officer", "RECORDS SECTION", "RECORDS", "records", "records123", "Officer", new Date().toISOString()]);
  } else {
    userSheet.getRange(1, 1, 1, expectedUserHeaders.length).setValues([expectedUserHeaders]).setFontWeight("bold").setBackground("#FEF7E0");
  }
  let setSheet = ss.getSheetByName("Settings");
  if (!setSheet) {
    setSheet = ss.insertSheet("Settings");
    setSheet.appendRow(["Key", "Value", "UpdatedAt"]);
    setSheet.getRange("A1:C1").setFontWeight("bold").setBackground("#FCE8E6");
  }

  purgeExpiredTrash(docSheet, logSheet);
  getAdminCode();
  getRouteOverrideCode();
  getOrCreateArchiveFolder();
  return ss.getUrl();
}

function generateNextOfficeTrackingId(ss, officeCodeInput) {
  const cleanCode = String(officeCodeInput || "DOC").trim().toUpperCase().replace(/[^A-Z0-9]/g, "") || "DOC";
  const now = new Date();
  const mm = String(now.getMonth() + 1).padStart(2, "0");
  const yyyy = String(now.getFullYear());
  const docSheet = ss.getSheetByName("Documents");
  let maxSeq = 0;
  if (docSheet && docSheet.getLastRow() > 1) {
    const ids = docSheet.getRange(2, 1, docSheet.getLastRow() - 1, 1).getValues();
    const regex = new RegExp("^" + cleanCode + "-\\d{2}-" + yyyy + "-(\\d+)$", "i");
    for (let i = 0; i < ids.length; i++) {
      const val = String(ids[i][0] || "").trim();
      const match = val.match(regex);
      if (match && match[1]) {
        const num = parseInt(match[1], 10);
        if (!isNaN(num) && num > maxSeq) maxSeq = num;
      }
    }
  }
  return cleanCode + "-" + mm + "-" + yyyy + "-" + String(maxSeq + 1).padStart(3, "0");
}

function doGet(e) {
  try {
    const sheetUrl = setupSheets();
    const ss = getDatabaseSpreadsheet();
    return jsonResponse({
      status: "success",
      sheetUrl: sheetUrl,
      documents: sheetToObjects(ss.getSheetByName("Documents")),
      logs: sheetToObjects(ss.getSheetByName("TrackingLogs")),
      users: sheetToObjects(ss.getSheetByName("Users")),
      settings: sheetToObjects(ss.getSheetByName("Settings")).reduce((acc, r) => { if (r.Key) acc[r.Key] = r.Value; return acc; }, {})
    });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  }
}

function doPost(e) {
  try {
    const sheetUrl = setupSheets();
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;
    const ss = getDatabaseSpreadsheet();
    const currentSecret = getAdminCode();
    const currentOverrideCode = getRouteOverrideCode();

    if (action === "verifyAdmin") {
      const isValid = (String(payload.adminCode) === String(currentSecret));
      return jsonResponse({ status: isValid ? "success" : "unauthorized", valid: isValid, sheetUrl: isValid ? sheetUrl : "" });
    }

    if (action === "loginUser") {
      const uname = String(payload.username || "").trim().toLowerCase();
      const pwd = String(payload.password || "").trim();
      const matched = sheetToObjects(ss.getSheetByName("Users")).find(u => String(u.Username || "").trim().toLowerCase() === uname && String(u.Password || "").trim() === pwd);
      if (!matched) return jsonResponse({ status: "unauthorized", message: "Invalid username or password." });
      return jsonResponse({
        status: "success",
        user: { id: String(matched.UserID || ""), name: String(matched.Name || ""), office: String(matched.Office || ""), officeCode: String(matched.OfficeCode || ""), username: String(matched.Username || ""), role: String(matched.Role || "Officer") }
      });
    }

    if (action === "verifyRouteOverride") {
      const isValid = (String(payload.overrideCode) === String(currentOverrideCode));
      return jsonResponse({ status: isValid ? "success" : "unauthorized", valid: isValid });
    }

    if (action === "changeAdminPassword") {
      if (String(payload.currentCode) !== String(currentSecret)) return jsonResponse({ status: "unauthorized", message: "Current passcode is incorrect." });
      PropertiesService.getScriptProperties().setProperty("ADMIN_SECRET_CODE", String(payload.newCode).trim());
      return jsonResponse({ status: "success" });
    }

    if (action === "changeRouteOverrideCode") {
      if (String(payload.adminCode) !== String(currentSecret)) return jsonResponse({ status: "unauthorized", message: "Admin passcode incorrect." });
      PropertiesService.getScriptProperties().setProperty("ROUTE_OVERRIDE_CODE", String(payload.newCode).trim());
      return jsonResponse({ status: "success" });
    }

    if (action === "saveSettings") {
      const setSheet = ss.getSheetByName("Settings");
      const items = payload.settings || {};
      const data = setSheet.getDataRange().getValues();
      Object.keys(items).forEach(key => {
        let foundRow = -1;
        for (let i = 1; i < data.length; i++) { if (data[i][0] === key) { foundRow = i + 1; break; } }
        if (foundRow > 0) { setSheet.getRange(foundRow, 2).setValue(items[key]); setSheet.getRange(foundRow, 3).setValue(new Date().toISOString()); }
        else { setSheet.appendRow([key, items[key], new Date().toISOString()]); }
      });
      return jsonResponse({ status: "success" });
    }

    if (action === "createDocument") {
      const d = payload.document;
      const docSheet = ss.getSheetByName("Documents");
      const finalTrackingId = d.officeCode ? generateNextOfficeTrackingId(ss, d.officeCode) : (d.id || generateNextOfficeTrackingId(ss, "DOC"));
      const metaJson = JSON.stringify({
        selectedTypes: d.selectedTypes || [], dvOptions: d.dvOptions || {}, fillIns: d.fillIns || {},
        routeOffices: d.routeOffices || [], currentRouteIndex: d.currentRouteIndex || 0, lastUpdatedMs: d.lastUpdatedMs || Date.now(), officeCode: d.officeCode || ""
      });
      docSheet.appendRow([finalTrackingId, d.title, d.startParty || "", d.type, d.priority, d.origin, d.destination || "", d.custom2 || "", d.remarks, d.status, d.currentOffice, d.createdBy, d.createdAt, metaJson, new Date().toISOString(), ""]);
      ss.getSheetByName("TrackingLogs").appendRow(["LOG-" + Date.now(), finalTrackingId, d.origin, "FORWARDED", d.createdBy, d.remarks + " (Route Started)", d.createdAt, ""]);
      return jsonResponse({ status: "success", trackingId: finalTrackingId });
    }

    if (action === "scanDocument") {
      const s = payload.scan;
      let uploadedDriveUrl = "";
      if (s.fileDataUrl && s.fileDataUrl.indexOf("base64,") !== -1) {
        const monthlyFolder = getOrCreateMonthlyArchiveFolder();
        const parts = s.fileDataUrl.split("base64,");
        const mimeType = parts[0].replace("data:", "").replace(";", "");
        const ext = mimeType.indexOf("pdf") !== -1 ? ".pdf" : ".png";
        const blob = Utilities.newBlob(Utilities.base64Decode(parts[1]), mimeType, s.trackingId + "_Completed" + ext);
        const file = monthlyFolder.createFile(blob);
        try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}
        uploadedDriveUrl = "https://drive.google.com/file/d/" + file.getId() + "/view?usp=sharing";
      }
      ss.getSheetByName("TrackingLogs").appendRow(["LOG-" + Date.now(), s.trackingId, s.office, s.actionStatus, s.handledBy, s.remarks, s.timestamp, uploadedDriveUrl]);
      const docSheet = ss.getSheetByName("Documents");
      const data = docSheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(s.trackingId)) {
          let meta = {};
          try { meta = JSON.parse(data[i][13] || "{}"); } catch (e) {}
          if (s.nextRouteIndex !== undefined) meta.currentRouteIndex = s.nextRouteIndex;
          meta.lastUpdatedMs = Date.now();
          docSheet.getRange(i + 1, 10).setValue(s.actionStatus);
          docSheet.getRange(i + 1, 11).setValue(s.office);
          docSheet.getRange(i + 1, 14).setValue(JSON.stringify(meta));
          docSheet.getRange(i + 1, 15).setValue(new Date().toISOString());
          if (uploadedDriveUrl) docSheet.getRange(i + 1, 16).setValue(uploadedDriveUrl);
          break;
        }
      }
      return jsonResponse({ status: "success", fileUrl: uploadedDriveUrl });
    }

    if (action === "addUser") {
      if (String(payload.adminCode) !== String(currentSecret)) return jsonResponse({ status: "unauthorized", message: "Invalid Admin Code" });
      const u = payload.user;
      const uSheet = ss.getSheetByName("Users");
      const uData = uSheet.getDataRange().getValues();
      const targetUname = String(u.username || "").trim().toLowerCase();
      for (let i = 1; i < uData.length; i++) {
        if (String(uData[i][0]) === String(u.id) || (targetUname && String(uData[i][4] || "").trim().toLowerCase() === targetUname)) {
          uSheet.getRange(i + 1, 1, 1, 7).setValues([[u.id, u.name, u.office, u.officeCode, u.username, u.password, u.role || "Officer"]]);
          return jsonResponse({ status: "success", updatedExisting: true });
        }
      }
      uSheet.appendRow([u.id, u.name, u.office, u.officeCode, u.username, u.password, u.role || "Officer", new Date().toISOString()]);
      return jsonResponse({ status: "success" });
    }

    if (action === "editUser") {
      if (String(payload.adminCode) !== String(currentSecret)) return jsonResponse({ status: "unauthorized", message: "Invalid Admin Code" });
      const u = payload.user;
      const uSheet = ss.getSheetByName("Users");
      const uData = uSheet.getDataRange().getValues();
      for (let i = 1; i < uData.length; i++) {
        if (String(uData[i][0]) === String(u.id)) {
          uSheet.getRange(i + 1, 2, 1, 7).setValues([[u.name, u.office, u.officeCode, u.username, u.password, u.role || "Officer", new Date().toISOString()]]);
          return jsonResponse({ status: "success" });
        }
      }
      uSheet.appendRow([u.id, u.name, u.office, u.officeCode, u.username, u.password, u.role || "Officer", new Date().toISOString()]);
      return jsonResponse({ status: "success" });
    }

    if (action === "deleteUser") {
      if (String(payload.adminCode) !== String(currentSecret)) return jsonResponse({ status: "unauthorized", message: "Invalid Admin Code" });
      const targetUserId = String(payload.userId || "");
      const uSheet = ss.getSheetByName("Users");
      const uData = uSheet.getDataRange().getValues();
      for (let i = uData.length - 1; i >= 1; i--) { if (String(uData[i][0]) === targetUserId) uSheet.deleteRow(i + 1); }
      return jsonResponse({ status: "success" });
    }

    if (action === "trashDocument") {
      if (String(payload.adminCode) !== String(currentSecret)) return jsonResponse({ status: "unauthorized", message: "Invalid Admin Code" });
      const targetId = String(payload.trackingId || "");
      const dSheet = ss.getSheetByName("Documents");
      const dData = dSheet.getDataRange().getValues();
      const nowYyyyMm = new Date().getFullYear() + "-" + String(new Date().getMonth() + 1).padStart(2, "0");
      for (let i = 1; i < dData.length; i++) {
        if (String(dData[i][0]) === targetId) {
          dSheet.getRange(i + 1, 10).setValue("TRASH");
          let meta = {};
          try { meta = JSON.parse(dData[i][13] || "{}"); } catch (e) {}
          meta.deletedMonth = nowYyyyMm;
          meta.previousStatus = dData[i][9];
          dSheet.getRange(i + 1, 14).setValue(JSON.stringify(meta));
          break;
        }
      }
      return jsonResponse({ status: "success" });
    }

    if (action === "recoverDocument") {
      if (String(payload.adminCode) !== String(currentSecret)) return jsonResponse({ status: "unauthorized", message: "Invalid Admin Code" });
      const targetId = String(payload.trackingId || "");
      const dSheet = ss.getSheetByName("Documents");
      const dData = dSheet.getDataRange().getValues();
      for (let i = 1; i < dData.length; i++) {
        if (String(dData[i][0]) === targetId) {
          let meta = {};
          try { meta = JSON.parse(dData[i][13] || "{}"); } catch (e) {}
          const prevStatus = meta.previousStatus || "IN TRANSIT";
          delete meta.deletedMonth;
          delete meta.previousStatus;
          dSheet.getRange(i + 1, 10).setValue(prevStatus);
          dSheet.getRange(i + 1, 14).setValue(JSON.stringify(meta));
          break;
        }
      }
      return jsonResponse({ status: "success" });
    }

    if (action === "deleteDocumentForever") {
      if (String(payload.adminCode) !== String(currentSecret)) return jsonResponse({ status: "unauthorized", message: "Invalid Admin Code" });
      const targetId = String(payload.trackingId || "");
      const dSheet = ss.getSheetByName("Documents");
      const dData = dSheet.getDataRange().getValues();
      for (let i = dData.length - 1; i >= 1; i--) { if (String(dData[i][0]) === targetId) dSheet.deleteRow(i + 1); }
      const lSheet = ss.getSheetByName("TrackingLogs");
      const lData = lSheet.getDataRange().getValues();
      for (let j = lData.length - 1; j >= 1; j--) { if (String(lData[j][1]) === targetId) lSheet.deleteRow(j + 1); }
      return jsonResponse({ status: "success" });
    }

    return jsonResponse({ status: "error", message: "Invalid action" });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  }
}

function sheetToObjects(sheet) {
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return [];
  const headers = data[0];
  return data.slice(1).map(row => { let obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; });
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
