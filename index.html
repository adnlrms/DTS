/**
 * DOCTRACK QR - GOOGLE SHEETS & GOOGLE DRIVE BACKEND (Code.gs)
 * Supports QR Routing, Bottleneck Tracking, and End-of-Process Google Drive Document Storage.
 */

const INITIAL_ADMIN_CODE = "ADMIN-2026";

function getAdminCode() {
  const props = PropertiesService.getScriptProperties();
  let code = props.getProperty("ADMIN_SECRET_CODE");
  if (!code) {
    code = INITIAL_ADMIN_CODE;
    props.setProperty("ADMIN_SECRET_CODE", code);
  }
  return code;
}

function getDatabaseSpreadsheet() {
  let ss = null;
  try { ss = SpreadsheetApp.getActiveSpreadsheet(); } catch (e) {}
  if (ss) return ss;

  const props = PropertiesService.getScriptProperties();
  const savedSheetId = props.getProperty("DATABASE_SHEET_ID");
  if (savedSheetId) {
    try {
      ss = SpreadsheetApp.openById(savedSheetId);
      if (ss) return ss;
    } catch (e) {}
  }

  ss = SpreadsheetApp.create("DocTrack QR - DepEd Agusan del Norte Database");
  props.setProperty("DATABASE_SHEET_ID", ss.getId());
  Logger.log("✅ Created new Google Sheet Database: " + ss.getUrl());
  return ss;
}

function getOrCreateArchiveFolder() {
  const props = PropertiesService.getScriptProperties();
  const savedFolderId = props.getProperty("ARCHIVE_FOLDER_ID");
  if (savedFolderId) {
    try {
      const f = DriveApp.getFolderById(savedFolderId);
      if (f) return f;
    } catch (e) {}
  }
  const folder = DriveApp.createFolder("DocTrack Completed Document Scans");
  try { folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}
  props.setProperty("ARCHIVE_FOLDER_ID", folder.getId());
  return folder;
}

function setupSheets() {
  const ss = getDatabaseSpreadsheet();

  let docSheet = ss.getSheetByName("Documents");
  if (!docSheet) {
    docSheet = ss.insertSheet("Documents");
    docSheet.appendRow([
      "TrackingID", "Title", "StartParty", "Type", "Priority",
      "OriginOffice", "DestinationOffice", "EndNote", "Remarks",
      "Status", "CurrentOffice", "CreatedBy", "CreatedAt", "MetaJSON",
      "LastUpdatedAt", "ScannedFileUrl"
    ]);
    docSheet.getRange("A1:P1").setFontWeight("bold").setBackground("#E8F0FE");
  } else if (docSheet.getLastColumn() < 16) {
    docSheet.getRange(1, 15).setValue("LastUpdatedAt");
    docSheet.getRange(1, 16).setValue("ScannedFileUrl");
  }

  if (!ss.getSheetByName("TrackingLogs")) {
    const logSheet = ss.insertSheet("TrackingLogs");
    logSheet.appendRow(["LogID", "TrackingID", "Office", "Action", "HandledBy", "Remarks", "Timestamp", "FileUrl"]);
    logSheet.getRange("A1:H1").setFontWeight("bold").setBackground("#E6F4EA");
  }

  if (!ss.getSheetByName("Users")) {
    const userSheet = ss.insertSheet("Users");
    userSheet.appendRow(["UserID", "Name", "Office", "Role", "CreatedAt"]);
    userSheet.getRange("A1:E1").setFontWeight("bold").setBackground("#FEF7E0");
    userSheet.appendRow(["USR-01", "Records Officer", "RECORDS SECTION", "Officer", new Date().toISOString()]);
  }

  if (!ss.getSheetByName("Settings")) {
    const setSheet = ss.insertSheet("Settings");
    setSheet.appendRow(["Key", "Value", "UpdatedAt"]);
    setSheet.getRange("A1:C1").setFontWeight("bold").setBackground("#FCE8E6");
  }

  const defaultSheet = ss.getSheetByName("Sheet1");
  if (defaultSheet && ss.getSheets().length > 1) {
    try { ss.deleteSheet(defaultSheet); } catch (e) {}
  }

  getAdminCode();
  getOrCreateArchiveFolder();
  Logger.log("✅ Database & Drive Folder Ready! Sheet URL: " + ss.getUrl());
  return ss.getUrl();
}

function doGet(e) {
  try {
    const sheetUrl = setupSheets();
    const ss = getDatabaseSpreadsheet();
    const docs = sheetToObjects(ss.getSheetByName("Documents"));
    const logs = sheetToObjects(ss.getSheetByName("TrackingLogs"));
    const users = sheetToObjects(ss.getSheetByName("Users"));
    const settingsRows = sheetToObjects(ss.getSheetByName("Settings"));
    const settings = {};
    settingsRows.forEach(r => { if (r.Key) settings[r.Key] = r.Value; });

    return jsonResponse({
      status: "success",
      sheetUrl: sheetUrl,
      documents: docs,
      logs: logs,
      users: users,
      settings: settings
    });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  }
}

function doPost(e) {
  try {
    const sheetUrl = setupSheets();
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;
    const ss = getDatabaseSpreadsheet();
    const currentSecret = getAdminCode();

    if (action === "verifyAdmin") {
      const isValid = (String(payload.adminCode) === String(currentSecret));
      return jsonResponse({ status: isValid ? "success" : "unauthorized", valid: isValid, sheetUrl: isValid ? sheetUrl : "" });
    }

    if (action === "changeAdminPassword") {
      if (String(payload.currentCode) !== String(currentSecret)) {
        return jsonResponse({ status: "unauthorized", message: "Current passcode is incorrect." });
      }
      PropertiesService.getScriptProperties().setProperty("ADMIN_SECRET_CODE", String(payload.newCode).trim());
      return jsonResponse({ status: "success" });
    }

    if (action === "saveSettings") {
      const setSheet = ss.getSheetByName("Settings");
      const items = payload.settings || {};
      const data = setSheet.getDataRange().getValues();
      Object.keys(items).forEach(key => {
        let foundRow = -1;
        for (let i = 1; i < data.length; i++) {
          if (data[i][0] === key) { foundRow = i + 1; break; }
        }
        if (foundRow > 0) {
          setSheet.getRange(foundRow, 2).setValue(items[key]);
          setSheet.getRange(foundRow, 3).setValue(new Date().toISOString());
        } else {
          setSheet.appendRow([key, items[key], new Date().toISOString()]);
        }
      });
      return jsonResponse({ status: "success" });
    }

    if (action === "createDocument") {
      const d = payload.document;
      const docSheet = ss.getSheetByName("Documents");
      const metaJson = JSON.stringify({
        selectedTypes: d.selectedTypes || [],
        dvOptions: d.dvOptions || {},
        fillIns: d.fillIns || {},
        lastUpdatedMs: d.lastUpdatedMs || Date.now()
      });
      docSheet.appendRow([
        d.id, d.title, d.startParty || "", d.type, d.priority,
        d.origin, d.destination, d.custom2 || "", d.remarks,
        d.status, d.currentOffice, d.createdBy, d.createdAt, metaJson,
        new Date().toISOString(), ""
      ]);

      const logSheet = ss.getSheetByName("TrackingLogs");
      logSheet.appendRow([
        "LOG-" + Date.now(), d.id, d.origin, "FORWARDED",
        d.createdBy, d.remarks + " (To: " + d.destination + ")", d.createdAt, ""
      ]);
      return jsonResponse({ status: "success", trackingId: d.id });
    }

    if (action === "scanDocument") {
      const s = payload.scan;
      let uploadedDriveUrl = "";

      if (s.fileDataUrl && s.fileDataUrl.indexOf("base64,") !== -1) {
        const folder = getOrCreateArchiveFolder();
        const parts = s.fileDataUrl.split("base64,");
        const contentType = parts[0].replace("data:", "").replace(";", "");
        const bytes = Utilities.base64Decode(parts[1]);
        const blob = Utilities.newBlob(bytes, contentType, s.trackingId + "_Completed_Scan.png");
        const file = folder.createFile(blob);
        try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (err) {}
        uploadedDriveUrl = "https://drive.google.com/uc?export=view&id=" + file.getId();
      }

      const logSheet = ss.getSheetByName("TrackingLogs");
      logSheet.appendRow([
        "LOG-" + Date.now(), s.trackingId, s.office,
        s.actionStatus, s.handledBy, s.remarks, s.timestamp, uploadedDriveUrl
      ]);

      const docSheet = ss.getSheetByName("Documents");
      const data = docSheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(s.trackingId)) {
          const newStatus = (s.actionStatus === "COMPLETED" || s.actionStatus === "RETURNED") ? s.actionStatus : "IN TRANSIT";
          docSheet.getRange(i + 1, 10).setValue(newStatus);
          docSheet.getRange(i + 1, 11).setValue(s.office);
          docSheet.getRange(i + 1, 15).setValue(new Date().toISOString());
          if (uploadedDriveUrl) {
            docSheet.getRange(i + 1, 16).setValue(uploadedDriveUrl);
          }
          break;
        }
      }
      return jsonResponse({ status: "success", fileUrl: uploadedDriveUrl });
    }

    if (action === "addUser") {
      if (String(payload.adminCode) !== String(currentSecret)) {
        return jsonResponse({ status: "unauthorized", message: "Invalid Admin Code" });
      }
      const u = payload.user;
      const userSheet = ss.getSheetByName("Users");
      userSheet.appendRow([u.id, u.name, u.office, u.role || "Officer", new Date().toISOString()]);
      return jsonResponse({ status: "success" });
    }

    if (action === "deleteUser") {
      if (String(payload.adminCode) !== String(currentSecret)) {
        return jsonResponse({ status: "unauthorized", message: "Invalid Admin Code" });
      }
      const targetUserId = String(payload.userId || "");
      const userSheet = ss.getSheetByName("Users");
      if (userSheet) {
        const userData = userSheet.getDataRange().getValues();
        for (let i = userData.length - 1; i >= 1; i--) {
          if (String(userData[i][0]) === targetUserId) {
            userSheet.deleteRow(i + 1);
          }
        }
      }
      return jsonResponse({ status: "success", deletedUserId: targetUserId });
    }

    if (action === "deleteDocument") {
      if (String(payload.adminCode) !== String(currentSecret)) {
        return jsonResponse({ status: "unauthorized", message: "Invalid Admin Code" });
      }
      const targetId = String(payload.trackingId || "");
      const docSheet = ss.getSheetByName("Documents");
      if (docSheet) {
        const docData = docSheet.getDataRange().getValues();
        for (let i = docData.length - 1; i >= 1; i--) {
          if (String(docData[i][0]) === targetId) {
            docSheet.deleteRow(i + 1);
          }
        }
      }
      const logSheet = ss.getSheetByName("TrackingLogs");
      if (logSheet) {
        const logData = logSheet.getDataRange().getValues();
        for (let j = logData.length - 1; j >= 1; j--) {
          if (String(logData[j][1]) === targetId) {
            logSheet.deleteRow(j + 1);
          }
        }
      }
      return jsonResponse({ status: "success", deletedId: targetId });
    }

    return jsonResponse({ status: "error", message: "Invalid action: " + action });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  }
}

function sheetToObjects(sheet) {
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return [];
  const headers = data[0];
  return data.slice(1).map(row => {
    let obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    return obj;
  });
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
